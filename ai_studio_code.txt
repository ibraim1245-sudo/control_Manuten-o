name: Gerar Instalador Windows
on: [push]
jobs:
  build:
    runs-on: windows-latest
    steps:
      - name: Checkout Código
        uses: actions/checkout@v3

      - name: Configurar Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.10'

      - name: Instalar Dependências
        run: |
          python -m pip install --upgrade pip
          pip install eel pyinstaller

      - name: Compilar EXE
        run: |
          python -m eel app.py web --onefile --noconsole --name "ControleFerramentas"

      - name: Upload do Executável
        uses: actions/upload-artifact@v3
        with:
          name: App-Manutencao-Sem-Senha
          path: dist/ControleFerramentas.exe
